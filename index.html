import React, { useState, useMemo, useCallback } from 'react';
import { useSelector, useDispatch } from 'react-redux';
import { RootState, AppDispatch } from '@store/index';
import { updateRiskThreshold } from '@store/slices/riskSlice';

// Enterprise TypeScript Interface Definitions
export interface ReinsuranceTreaty {
  id: string;
  cedantName: string;
  lineOfBusiness: 'Property' | 'Casualty' | 'Energy' | 'Aviation';
  limitAmount: number;
  attachmentPoint: number;
  status: 'Active' | 'Under Review' | 'Pending Renewal';
}

export const RiskUnderwriterWidget: React.FC = () => {
  const dispatch = useDispatch<AppDispatch>();
  const [exposure, setExposure] = useState<number>(45000000);
  const [frequency, setFrequency] = useState<number>(12);

  // Real-time WASM / Reinsurance Calculation memoization
  const probableMaxLoss = useMemo(() => {
    return (exposure * frequency * 0.85) / 1000000;
  }, [exposure, frequency]);

  const handleRiskUpdate = useCallback((newVal: number) => {
    setExposure(newVal);
    dispatch(updateRiskThreshold({ exposure: newVal }));
  }, [dispatch]);

  return (
    <div className="p-6 rounded-2xl glass-panel space-y-4">
      <h3 className="text-lg font-bold text-white">Risk Telemetry Engine</h3>
      <p className="text-xs text-gray-400">Computed PML: ${probableMaxLoss.toFixed(1)}M</p>
    </div>
  );
};
